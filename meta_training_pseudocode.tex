\section{Overview}

MetaTrainer is a two-phase training framework combining First-Order Model-Agnostic Meta-Learning (FOMAML) with entropy-based curriculum expansion and PPO-based task-specific fine-tuning. The algorithm trains a shared policy across multiple tasks via meta-learning, then optimizes task-specific policies through supervised fine-tuning.

\begin{algorithm}[h]
\caption{MetaTrainer: Complete Two-Phase Training Pipeline}
\label{alg:metatrainer}
\begin{algorithmic}[1]

\Require{
\begin{itemize}
	\item Shared policy network $\pi_\theta$ with learnable parameters $\theta$
	\item Task-specific environment with task list $\mathcal{T} = \{t_1, \ldots, t_K\}$
	\item Meta-learning configuration $\mathcal{C}_{\text{meta}}$ (epochs, batches, entropy threshold)
	\item Fine-tuning configuration $\mathcal{C}_{\text{tune}}$ (iterations, rollout length, PPO settings)
	\item Control flags $\text{do\_meta\_learning}, \text{do\_fine\_tuning}$
\end{itemize}
}

\Ensure{
\begin{itemize}
	\item Converged meta-policy $\pi_{\theta^*}$ (from Phase 1)
	\item Per-task fine-tuned policies $\{\pi_{\theta_1^*}, \ldots, \pi_{\theta_K^*}\}$ (from Phase 2)
	\item Training summary: best objectives, convergence logs, checkpoints
\end{itemize}
}

\State \textbf{Initialize:} $\text{total\_updates} \leftarrow 0$, $\text{best\_objective} \leftarrow -\infty$

\State \textbf{Log training header with configuration}

\State \textbf{// Phase 1: Meta-Learning with Curriculum Expansion}
\If{\text{do\_meta\_learning}}
	\State $(\theta^*, \text{meta\_summary}) \leftarrow \text{PHASE\_META\_LEARN}(\pi_\theta, \mathcal{C}_{\text{meta}})$
	\Comment{Algorithm \ref{alg:phase1}}
	\State $\text{best\_objective} \leftarrow \max(\text{best\_objective}, \text{meta\_summary.best\_obj})$
	\State $\text{total\_updates} \leftarrow \text{total\_updates} + \text{meta\_summary.num\_updates}$
	\State $\pi_\theta \leftarrow \theta^*$ \Comment{Update policy with meta-learned weights}
\Else
	\State \textbf{Log:} "Meta-learning skipped; using random or pretrained initialization"
\EndIf

\State \textbf{// Phase 2: Per-Task Fine-Tuning}
\If{\text{do\_fine\_tuning}}
	\State $(\text{task\_policies}, \text{tune\_summary}) \leftarrow \text{PHASE\_FINE\_TUNE}(\pi_\theta, \mathcal{C}_{\text{tune}})$
	\Comment{Algorithm \ref{alg:phase2}}
	\State $\text{best\_objective} \leftarrow \max(\text{best\_objective}, \text{tune\_summary.best\_obj})$
	\State $\text{total\_updates} \leftarrow \text{total\_updates} + \text{tune\_summary.num\_updates}$
\Else
	\State \textbf{Log:} "Fine-tuning skipped; meta-learning complete"
\EndIf

\State \textbf{Log training footer with summary}
\State \textbf{return} $(\pi_{\theta^*}, \text{task\_policies}, \text{total\_updates}, \text{best\_objective})$

\end{algorithmic}
\end{algorithm}

\begin{algorithm}[h]
\caption{PHASE\_META\_LEARN: FOMAML with Entropy-Based Curriculum}
\label{alg:phase1}
\begin{algorithmic}[1]

\Require{
\begin{itemize}
	\item Policy $\pi_\theta$, Config $\mathcal{C}_{\text{meta}}$
\end{itemize}
}

\Ensure{
\begin{itemize}
	\item Converged meta-policy parameters $\theta^*$, Summary statistics
\end{itemize}
}

\State \textbf{Initialize:} $\text{active\_tasks} \leftarrow \{t_1\}$ (start with easiest task)
\State \textbf{Initialize:} $\text{best\_obj} \leftarrow -\infty$, $\text{patience} \leftarrow 0$, $\text{curriculum\_counter} \leftarrow 0$

\For{epoch $e = 1$ to $\mathcal{C}_{\text{meta}}.\text{num\_epochs}$}

	\State \textbf{Initialize:} $\text{epoch\_losses} \leftarrow []$, $\text{epoch\_metrics} \leftarrow \{\}$

	\For{batch $b = 1$ to $\mathcal{C}_{\text{meta}}.\text{batches\_per\_epoch}$}

		\State \textbf{// Compute meta-loss via FOMAML inner-outer loop}
		\State $(\text{task\_losses}, \text{metrics}) \leftarrow \text{COMPUTE\_FOMAML\_LOSSES}(\pi_\theta, \text{active\_tasks})$

		\State \textbf{// Meta-update: aggregate losses and gradient step}
		\State $\mathcal{L}_{\text{meta}} \leftarrow \frac{1}{|\text{active\_tasks}|} \sum_{\tau} \mathcal{L}_\tau$
		\State $\pi_\theta \leftarrow \text{OPTIMIZER\_STEP}(\pi_\theta, \mathcal{L}_{\text{meta}}, \text{learning\_rate})$

		\State \textbf{// Accumulate epoch statistics}
		\State Merge metrics into $\text{epoch\_metrics}$
		\State $\text{epoch\_losses} \leftarrow \text{epoch\_losses} + \text{task\_losses}$

		\State \textbf{// Curriculum expansion via entropy threshold}
		\State $\text{curriculum\_counter} \leftarrow \text{curriculum\_counter} + 1$
		\If{$\text{curriculum\_counter} \geq \mathcal{C}_{\text{meta}}.\text{check\_interval}$}
			\State $\text{curriculum\_counter} \leftarrow 0$
			\State $\text{max\_entropy} \leftarrow \max_\tau (\text{metrics}[\tau].\text{entropy})$
			\If{$\text{max\_entropy} < \mathcal{C}_{\text{meta}}.\text{entropy\_threshold}$ \textbf{and} $|\text{active\_tasks}| < |\mathcal{T}|$}
				\State $\text{next\_idx} \leftarrow |\text{active\_tasks}|$
				\State $\text{active\_tasks} \leftarrow \text{active\_tasks} \cup \{t_{\text{next\_idx}}\}$
				\State \textbf{Log:} "Curriculum expansion: added task", $|\text{active\_tasks}|$ tasks active
			\EndIf
		\EndIf

	\EndFor

	\State \textbf{// Epoch-level evaluation and early stopping}
	\If{$e \mod \mathcal{C}_{\text{meta}}.\text{eval\_interval} = 0$}
		\State $\text{eval\_obj} \leftarrow \text{EVALUATE}(\pi_\theta, \text{median\_difficulty\_task})$
		\If{$\text{eval\_obj} > \text{best\_obj} + \mathcal{C}_{\text{meta}}.\text{min\_delta}$}
			\State $\text{best\_obj} \leftarrow \text{eval\_obj}$
			\State $\text{patience} \leftarrow 0$
			\State \textbf{Save checkpoint:} meta\_best
		\Else
			\State $\text{patience} \leftarrow \text{patience} + 1$
		\EndIf
		\If{$\text{patience} \geq \mathcal{C}_{\text{meta}}.\text{patience}$}
			\State \textbf{Break} (early stopping)
		\EndIf
	\EndIf

	\State \textbf{Log:} epoch, loss, num\_active\_tasks, best\_obj

\EndFor

\State \textbf{return} $(\pi_\theta, \{\text{epochs}, \text{best\_obj}, \text{final\_num\_tasks}\})$

\end{algorithmic}
\end{algorithm}

% FOMAML INNER-OUTER LOOP
% ============================================================================

\begin{algorithm}[h]
\caption{COMPUTE\_FOMAML\_LOSSES: Inner-Loop Adaptation + Outer-Loop Evaluation}
\label{alg:fomaml}
\begin{algorithmic}[1]

\Require{
\begin{itemize}
	\item Policy $\pi_\theta$, Active task set $\text{active\_tasks}$
	\item Hyperparameters: inner LR $\alpha_{\text{in}}$, clip\_eps, value\_coef, entropy\_coef
\end{itemize}
}

\Ensure{
\begin{itemize}
	\item Query losses for all tasks, task metrics (entropy, improvements)
\end{itemize}
}

\State \textbf{Initialize:} $\text{query\_losses} \leftarrow []$, $\text{task\_metrics} \leftarrow \{\}$

\For{each task $\tau \in \text{active\_tasks}$}

	\State \textbf{// Inner Loop: Collect support set and compute adapted parameters}
	\State $\theta_{\text{old}} \leftarrow \text{clone}(\pi_\theta)$ \Comment{Clone for this task}
	\State $\text{batch\_supp} \leftarrow \text{COLLECT\_EPISODE}(\pi_{\theta_{\text{old}}}, \tau)$ \Comment{One complete episode}
	\State $\mathcal{L}_{\text{supp}} \leftarrow \text{COMPUTE\_PPO\_LOSS}(\text{batch\_supp}, \pi_{\theta_{\text{old}}})$
	\State $\nabla_\theta \mathcal{L}_{\text{supp}} \leftarrow \text{BACKWARD}(\mathcal{L}_{\text{supp}})$
	\State $\theta_{\text{adapted}} \leftarrow \theta_{\text{old}} - \alpha_{\text{in}} \cdot \nabla_\theta \mathcal{L}_{\text{supp}}$ \Comment{FOMAML step}

	\State \textbf{// Outer Loop: Collect query set and evaluate with adapted parameters}
	\State $\text{batch\_query} \leftarrow \text{COLLECT\_EPISODE}(\pi_{\theta_{\text{old}}}, \tau)$ \Comment{Another episode from same task}
	\State $\mathcal{L}_{\text{query}} \leftarrow \text{COMPUTE\_PPO\_LOSS}(\text{batch\_query}, \pi, \theta_{\text{adapted}})$
	\Comment{Via functional\_call for gradient flow}

	\State \textbf{// Metrics for curriculum and monitoring}
	\State $\text{entropy} \leftarrow \text{mean}(\text{batch\_query.entropies})$
	\State $\text{improvement} \leftarrow \mathcal{L}_{\text{supp}} - \mathcal{L}_{\text{query}}$

	\State $\text{query\_losses}.\text{append}(\mathcal{L}_{\text{query}})$
	\State $\text{task\_metrics}[\tau] \leftarrow \{\text{supp\_loss}, \text{query\_loss}, \text{improvement}, \text{entropy}\}$

\EndFor

\State \textbf{return} $(\text{query\_losses}, \text{task\_metrics})$

\end{algorithmic}
\end{algorithm}

\begin{algorithm}[h]
\caption{COLLECT\_EPISODE: Single Episode with Advantage Estimation}
\label{alg:collect_ep}
\begin{algorithmic}[1]

\Require{
\begin{itemize}
	\item Policy $\pi_\theta$, Task $\tau$
	\item GAE parameters: $\gamma$ (discount), $\lambda$ (GAE factor)
\end{itemize}
}

\Ensure{
\begin{itemize}
	\item Trajectory batch: obs, actions, log\_probs, advantages, returns, entropies
\end{itemize}
}

\State \textbf{Initialize:} Empty trajectory buffers, $(s, \text{info}) \leftarrow \text{env.reset}(\tau)$

\State \textbf{// Rollout: execute policy until episode termination}
\While{\textbf{True}}
	\State $s_t \leftarrow \text{to\_tensor}(s)$, $\text{mask}_t \leftarrow \text{to\_tensor}(\text{info.action\_mask})$
	\State $(a_t, \log p_t, v_t, h_t) \leftarrow \pi_\theta.\text{act}(s_t, \text{mask}_t, \text{deterministic}=\text{False})$
	\State $(s', r_t, \text{done}, \text{truncated}, \text{info}) \leftarrow \text{env.step}(a_t)$
	\State \textbf{Buffer:} $(s_t, a_t, \log p_t, v_t, h_t, r_t, \text{done})$
	\State $s \leftarrow s'$
	\If{\text{done} or \text{truncated} or not $\text{info.action\_mask}$.any()}
		\State \textbf{break}
	\EndIf
\EndWhile

\State \textbf{// Bootstrap value at trajectory end}
\If{\text{truncated} and $\text{info.action\_mask}$.any()}
	\State $v_{\text{boot}} \leftarrow \pi_\theta.\text{get\_value}(\text{to\_tensor}(s))$
\Else
	\State $v_{\text{boot}} \leftarrow 0$
\EndIf

\State \textbf{// Compute Generalized Advantage Estimate}
\State $(A_t, G_t) \leftarrow \text{COMPUTE\_GAE}(R, V, D, v_{\text{boot}}, \gamma, \lambda)$
\State \textbf{Normalize advantages:} $A_t \leftarrow (A_t - \mu) / (\sigma + \epsilon)$

\State \textbf{return} $\{\text{states}, \text{actions}, \text{logprobs}, A_t, G_t, \text{entropies}\}$

\end{algorithmic}
\end{algorithm}

\begin{algorithm}[h]
\caption{PHASE\_FINE\_TUNE: Per-Task PPO with Vectorized Environments}
\label{alg:phase2}
\begin{algorithmic}[1]

\Require{
\begin{itemize}
	\item Meta-learned policy $\pi_\theta$ (or random init if Phase 1 skipped)
	\item Config $\mathcal{C}_{\text{tune}}$ (iterations, rollout\_length, batch\_size, PPO settings)
\end{itemize}
}

\Ensure{
\begin{itemize}
	\item Per-task fine-tuned policies, Summary with per-task best objectives
\end{itemize}
}

\State \textbf{Initialize:} $\text{vec\_env} \leftarrow \text{MAKE\_VECTORIZED\_ENV}(\mathcal{C}_{\text{tune}}.\text{batch\_size})$
\State \textbf{Initialize:} $\text{task\_policies} \leftarrow \{\}$, $\text{best\_obj} \leftarrow -\infty$

\For{each task $\tau \in \mathcal{T}$}

	\State \textbf{Initialize task:} $\text{task\_best} \leftarrow -\infty$, $\text{patience} \leftarrow 0$
	\State Create task-specific policy: $\pi_\tau \leftarrow \text{COPY}(\pi_\theta)$

	\For{iteration $i = 1$ to $\mathcal{C}_{\text{tune}}.\text{num\_iterations}$}

		\State \textbf{// Collect rollouts from parallel workers}
		\State $\text{vec\_env.retask}(\tau)$
		\State $\text{batch} \leftarrow \text{COLLECT\_ROLLOUTS}(\pi_\tau, \text{vec\_env}, \mathcal{C}_{\text{tune}}.\text{rollout\_length})$

		\State \textbf{// PPO updates: multiple epochs on collected data}
		\For{ppo\_epoch $= 1$ to $\mathcal{C}_{\text{tune}}.\text{ppo\_epochs}$}

			\For{each environment $e = 1$ to $\mathcal{C}_{\text{tune}}.\text{batch\_size}$}
				\State \textbf{// Get env-specific timesteps and create mini-batches}
				\State $\text{env\_indices} \leftarrow$ all timesteps from environment $e$
				\State \textbf{Shuffle} env\_indices
				\For{minibatch of size $\mathcal{C}_{\text{tune}}.\text{minibatch\_size}$}
					\State $\text{mb\_batch} \leftarrow$ extract observations, actions, advantages, returns
					\State $(\mathcal{L}_\text{policy}, \mathcal{L}_\text{value}, \mathcal{L}_\text{entropy}) \leftarrow \text{COMPUTE\_PPO\_LOSS}(\text{mb\_batch}, \pi_\tau)$
					\State $\pi_\tau \leftarrow \text{OPTIMIZER\_STEP}(\pi_\tau, \mathcal{L}_\text{total}, \alpha_\text{tune})$
				\EndFor
			\EndFor

		\EndFor

		\State \textbf{// Evaluate and checkpoint}
		\If{$i \mod \mathcal{C}_{\text{tune}}.\text{eval\_interval} = 0$}
			\State $\text{eval\_obj} \leftarrow \text{EVALUATE}(\pi_\tau, \tau)$
			\If{$\text{eval\_obj} > \text{task\_best} + \mathcal{C}_{\text{tune}}.\text{min\_delta}$}
				\State $\text{task\_best} \leftarrow \text{eval\_obj}$, $\text{patience} \leftarrow 0$
				\State \textbf{Save:} tune\_best\_$\tau$
			\Else
				\State $\text{patience} \leftarrow \text{patience} + 1$
				\If{$\text{patience} \geq \mathcal{C}_{\text{tune}}.\text{patience}$}
					\State \textbf{break}
				\EndIf
			\EndIf
		\EndIf

		\If{$i \mod \mathcal{C}_{\text{tune}}.\text{checkpoint\_interval} = 0$}
			\State \textbf{Save:} tune\_$\tau$\_iter\_$i$
		\EndIf

	\EndFor

	\State $\text{task\_policies}[\tau] \leftarrow \pi_\tau$
	\State $\text{best\_obj} \leftarrow \max(\text{best\_obj}, \text{task\_best})$

\EndFor

\State $\text{vec\_env.close}()$
\State \textbf{return} $(\text{task\_policies}, \{\text{num\_tasks}, \text{best\_obj}\})$

\end{algorithmic}
\end{algorithm}

\begin{algorithm}[h]
\caption{COLLECT\_ROLLOUTS: Parallel Rollout with Per-Environment Advantages}
\label{alg:collect_vec}
\begin{algorithmic}[1]

\Require{
\begin{itemize}
	\item Policy $\pi_\theta$, Vectorized env with $N$ workers
	\item Rollout length $T$, GAE params $\gamma, \lambda$
\end{itemize}
}

\Ensure{
\begin{itemize}
	\item Batch with $N \times T$ timesteps, per-env advantages and returns
\end{itemize}
}

\State \textbf{Initialize:} Buffers, $(s_\text{list}, \text{info\_list}) \leftarrow \text{vec\_env.reset}()$

\State \textbf{// Collect T timesteps from all N environments}
\For{$t = 1$ to $T$}
	\State $s_t \leftarrow \text{to\_tensor}(s_\text{list})$, $\text{mask}_t \leftarrow \text{to\_tensor}(\text{info\_list})$
	\State $(a_t, \log p_t, v_t, h_t) \leftarrow \pi_\theta.\text{act\_batch}(s_t, \text{mask}_t)$
	\State $(s_\text{list}, r_t, \text{done}, \text{trunc}, \text{info\_list}) \leftarrow \text{vec\_env.step}(a_t)$
	\State \textbf{Buffer timestep across all envs}
\EndFor

\State \textbf{// Bootstrap values for all envs}
\State $(\_,\_,v_\text{boot},\_) \leftarrow \pi_\theta.\text{act\_batch}(\text{to\_tensor}(s_\text{list}), \cdots)$
\State Mask out $v_\text{boot}$ for terminated episodes

\State \textbf{// Compute per-environment GAE}
\For{each environment $e$}
	\State $(A_e, G_e) \leftarrow \text{COMPUTE\_GAE}(r_e, v_e, d_e, v_\text{boot}[e], \gamma, \lambda)$
	\State Buffer $A_e, G_e$
\EndFor

\State \textbf{// Flatten and normalize across all envs}
\State $(A, G) \leftarrow \text{flatten\_and\_normalize}(\text{all per-env advantages/returns})$

\State \textbf{return} $\{\text{states}, \text{actions}, \text{logprobs}, A, G, \text{entropies}\}$

\end{algorithmic}
\end{algorithm}

% ============================================================================
% PPO LOSS COMPUTATION
% ============================================================================

\begin{algorithm}[h]
\caption{COMPUTE\_PPO\_LOSS: Policy Clipping + Value + Entropy}
\label{alg:ppo_loss}
\begin{algorithmic}[1]

\Require{
\begin{itemize}
	\item Batch: observations, actions, old\_logprobs, advantages, returns, masks
	\item Hyperparameters: $\epsilon$ (clip), $c_v$ (value coef), $c_e$ (entropy coef)
\end{itemize}
}

\Ensure{
\begin{itemize}
	\item Scalar total loss, component losses (for logging)
\end{itemize}
}

\State \textbf{// Forward pass}
\State $(\text{logits}, \text{values}) \leftarrow \pi_\theta(\text{observations}, \text{masks})$
\State $\log p_\text{new} \leftarrow \log\_\text{softmax}(\text{logits})[\text{actions}]$

\State \textbf{// PPO policy loss: clipped surrogate objective}
\State $r_t \leftarrow \exp(\log p_\text{new} - \text{old\_logprobs})$ \Comment{probability ratio}
\State $\text{surr1} \leftarrow r_t \times A_t$
\State $\text{surr2} \leftarrow \text{clip}(r_t, 1-\epsilon, 1+\epsilon) \times A_t$
\State $\mathcal{L}_\text{policy} \leftarrow -\text{mean}(\min(\text{surr1}, \text{surr2}))$

\State \textbf{// Value loss: MSE}
\State $\mathcal{L}_\text{value} \leftarrow \text{MSE}(\text{values}, \text{returns})$

\State \textbf{// Entropy loss: regularization for exploration}
\State $H \leftarrow -\sum_a \pi(a) \log \pi(a)$ \Comment{policy entropy}
\State $\mathcal{L}_\text{entropy} \leftarrow -\text{mean}(H)$ \Comment{negative to maximize entropy}

\State \textbf{// Combined loss}
\State $\mathcal{L}_\text{total} \leftarrow \mathcal{L}_\text{policy} + c_v \mathcal{L}_\text{value} + c_e \mathcal{L}_\text{entropy}$

\State \textbf{return} $(\mathcal{L}_\text{total}, \mathcal{L}_\text{policy}, \mathcal{L}_\text{value}, \mathcal{L}_\text{entropy})$

\end{algorithmic}
\end{algorithm}

\section{Mathematical Formulation}

\subsection{Generalized Advantage Estimation (GAE)}

GAE computes a bias-variance tradeoff between temporal difference and Monte Carlo returns:

\begin{equation}
\hat{A}_t = \sum_{l=0}^{T-t-1} (\gamma \lambda)^l \delta_{t+l}, \quad \text{where} \quad \delta_t = r_t + \gamma V(s_{t+1}) - V(s_t)
\end{equation}

\noindent
The TD residual $\delta_t$ is discounted and accumulated. Parameter $\lambda \in [0,1]$ controls bias-variance: $\lambda=0$ is pure TD, $\lambda=1$ is Monte Carlo.

\subsection{FOMAML Formulation}

\subsubsection{Inner Loop (Support Set)}
Compute task-specific gradient without stepping:
\begin{equation}
\mathcal{L}_\tau^{\text{supp}} = \text{PPO\_Loss}(\text{batch\_support}, \pi_\theta)
\end{equation}

\subsubsection{Adapted Parameters}
First-order approximation to task-optimal parameters:
\begin{equation}
\theta_\tau^{\text{adapted}} = \theta - \alpha_{\text{inner}} \nabla_\theta \mathcal{L}_\tau^{\text{supp}}
\end{equation}

\subsubsection{Outer Loop (Query Set)}
Evaluate loss with adapted parameters, backprop through adaptation:
\begin{equation}
\mathcal{L}_\tau^{\text{query}} = \text{PPO\_Loss}(\text{batch\_query}, \pi, \theta_\tau^{\text{adapted}})
\end{equation}

\subsubsection{Meta-Gradient}
Aggregate query losses across tasks:
\begin{equation}
\mathcal{L}_{\text{meta}} = \frac{1}{|\mathcal{T}|} \sum_{\tau \in \mathcal{T}} \mathcal{L}_\tau^{\text{query}}
\end{equation}

\subsection{PPO Loss}

Combined objective balancing policy improvement, value estimation, and exploration:

\begin{equation}
\mathcal{L}_{\text{PPO}} = \underbrace{-\mathbb{E}\left[\min\left(r_t \hat{A}_t, \text{clip}(r_t, 1-\epsilon, 1+\epsilon)\hat{A}_t\right)\right]}_{\text{Policy}} + c_v \underbrace{\text{MSE}(V(s), G_t)}_{\text{Value}} + c_e \underbrace{(-H(\pi))}_{\text{Entropy}}
\end{equation}

\noindent
where $r_t = \pi_\theta(a_t|s_t) / \pi_\text{old}(a_t|s_t)$ is the probability ratio.

\subsection{Entropy-Based Curriculum}

Tasks are scheduled dynamically. When policy entropy falls below threshold on current tasks, next task is activated:

\begin{equation}
\text{Activate}(t_{k+1}) \quad \Leftarrow \quad H(\pi_\theta(\cdot|s)) < H_{\text{threshold}}
\end{equation}

This prevents premature convergence on simple tasks and structures learning from easier to harder problems.

\section{Algorithm Complexity}

\subsection{Time Complexity}

\textbf{Meta-Learning Phase:}
\begin{equation}
O(E_{\text{meta}} \cdot B_{\text{epoch}} \cdot |\mathcal{T}|_{\text{active}} \cdot (2 T_{\text{ep}} + n_{\text{forward}} + 2 n_{\text{backward}}))
\end{equation}

\begin{itemize}
	\item $E_{\text{meta}}$ epochs of outer-loop iterations
	\item $B_{\text{epoch}}$ batches per epoch (FOMAML steps per epoch)
	\item $|\mathcal{T}|_{\text{active}}$ active tasks (grows gradually via curriculum)
	\item $2 T_{\text{ep}}$ for support + query episodes (serial collection)
	\item $n_{\text{forward}}, n_{\text{backward}}$ for network forward/backward passes
\end{itemize}

\textbf{Fine-Tuning Phase:}
\begin{equation}
O(|\mathcal{T}| \cdot I_{\text{tune}} \cdot B_{\text{size}} \cdot T_{\text{roll}} \cdot E_{\text{ppo}} \cdot \lceil \frac{B_{\text{size}} \cdot T_{\text{roll}}}{M} \rceil)
\end{equation}

\begin{itemize}
	\item $|\mathcal{T}|$ total tasks (sequential per-task training)
	\item $I_{\text{tune}}$ iterations per task
	\item $B_{\text{size}}$ parallel environments (vectorized)
	\item $T_{\text{roll}}$ rollout length per iteration
	\item $E_{\text{ppo}}$ PPO epochs per rollout
	\item $M$ mini-batch size
\end{itemize}

\subsection{Space Complexity}

\begin{itemize}
	\item \textbf{Model parameters:} $O(n_{\text{params}})$ (typically $10^6$--$10^7$)
	\item \textbf{Meta-phase buffers:} $O(|\mathcal{T}|_{\text{active}} \cdot 2T_{\text{ep}})$ (support + query per task)
	\item \textbf{Fine-tune buffers:} $O(B_{\text{size}} \cdot T_{\text{roll}})$ (cleared per iteration)
	\item \textbf{Vectorized env states:} $O(B_{\text{size}} \cdot |\text{state}|)$ (worker states)
\end{itemize}

\subsection{Computational Advantages}

\begin{enumerate}
	\item \textbf{Curriculum expansion:} Active task set grows gradually, reducing early computational cost
	\item \textbf{FOMAML:} First-order gradients only (no Hessian); single backward pass per task
	\item \textbf{Vectorized collection:} Parallel worker speedup in fine-tuning phase (wall-clock time)
	\item \textbf{Per-environment GAE:} On-the-fly computation avoids materializing global buffers
	\item \textbf{functional\_call:} Efficient adapted parameter evaluation without module cloning
\end{enumerate}
